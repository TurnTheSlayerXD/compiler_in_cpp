_Bool h_word_token(Queue *q, Token t, Tokenizer *tokenizer) {
    (void) tokenizer;

    if (t.type != T_CUSTOM_WORD) {
        return 0;
    }

    if (node_type_at(q, -1) == EXPR) {
        AstNode *expr = node_at(q, -1);
        while (((expr)->_.Expr).rhs) {
            if (!isinstance(((expr)->_.Expr).rhs, EXPR)) {
                break;
            }
            expr = ((expr)->_.Expr).rhs;
        }
        if (((expr)->_.Expr).rhs) {
            fprintf((__acrt_iob_func(2)), "Failed to find place for token at ""Tag pos:%d,row:%d,col:%d ""\n", (t.tag.pos),(t.tag.row),(t.tag.col));
            return 0;
        }

        ((expr)->_.Expr).rhs = AstNode_new_Word(t.text, t.tag);
        return 1;
    }

    push_queue(q, AstNode_new_Word(t.text, t.tag));
    return 1;
}

_Bool h_op_token(Queue *q, Token t, Tokenizer *tokenizer) {
    (void) tokenizer;
    if (!is_op_token(t)) {
        return 0;
    }

    if (node_type_at(q, -1) == EXPR) {
        AstNode *prev_expr = node_at(q, -1);
        AstNode *new_expr = AstNode_new_Expr(t, t.tag);
        if (node_cmp_expr(new_expr, prev_expr) < 0) {
            pop_queue(q);
            ((new_expr)->_.Expr).lhs = prev_expr;
            push_queue(q, new_expr);
        }
        else {
            AstNode *parent = prev_expr;
            while (((prev_expr)->_.Expr).rhs && isinstance(((prev_expr)->_.Expr).rhs, EXPR) && node_cmp_expr(new_expr, prev_expr) >= 0) {

                parent = prev_expr;
                prev_expr = ((prev_expr)->_.Expr).rhs;
            }

            if (parent == prev_expr) {
                if (node_cmp_expr(new_expr, parent) > 0) {
                    ((new_expr)->_.Expr).lhs = ((prev_expr)->_.Expr).rhs;
                    ((parent)->_.Expr).rhs = new_expr;
                }
                else {
                    pop_queue(q);
                    ((new_expr)->_.Expr).lhs = parent;
                    push_queue(q, new_expr);
                }

                return 1;
            }

            if (!((prev_expr)->_.Expr).rhs) {
                ((void)( (!!(0 && "No place for op node insertion")) || (_wassert(L"false && \"No place for op node insertion\"", L".\\ast.c", (unsigned)(416)), 0)) );
            }


            if (node_cmp_expr(new_expr, prev_expr) > 0) {
                ((new_expr)->_.Expr).lhs = ((prev_expr)->_.Expr).rhs;
                ((prev_expr)->_.Expr).rhs = new_expr;
            }

            else if (node_cmp_expr(new_expr, prev_expr) <= 0) {
                ((parent)->_.Expr).rhs = new_expr;
                ((new_expr)->_.Expr).lhs = prev_expr;
            }
        }
        return 1;
    }
    if (node_type_at(q, -1) == START_BRACKET) {
        push_queue(q, AstNode_new_Expr(t, t.tag));
        return 1;
    }
    if (node_type_at(q, -1) == WORD || node_type_at(q, -1) == CALL_EXPR) {
        AstNode *expr = AstNode_new_Expr(t, t.tag);
        AstNode *word = pop_queue(q);
        ((expr)->_.Expr).lhs = word;
        push_queue(q, expr);
        return 1;
    }

    push_queue(q, AstNode_new_Expr(t, t.tag));

    return 1;
}


_Bool is_leaf_type(NodeType n) {
    return n == CALL_EXPR || n == WORD || n == BRACE_EXPR;
}

_Bool h_brace_token(Queue *q, Token t, Tokenizer *tokenizer) {
    (void) tokenizer;

    switch (t.type) {
        case T_L_BR: case T_R_BR: break;
        default: return 0;
    }

    if (t.type == T_L_BR) {

        if (is_leaf_type(node_type_at(q, -1))) {
            return 0;
        }

        if (node_type_at(q, -1) == EXPR) {
            AstNode *expr = node_at(q, -1);
            while (((expr)->_.Expr).rhs && isinstance(((expr)->_.Expr).rhs, EXPR)) {
                expr = ((expr)->_.Expr).rhs;
            }
            if (((expr)->_.Expr).rhs) {
                return 0;
            }
        }

        push_queue(q, AstNode_new(START_BRACKET, t.tag));
        return 1;
    }
    if (t.type == T_R_BR && node_type_at(q, -2) == START_BRACKET) {

        AstNode *inside_brace_node = pop_queue(q);

        if (isinstance(inside_brace_node, EXPR)) {
            inside_brace_node->type = BRACE_EXPR;
        }

        pop_queue(q);

        if (node_type_at(q, -1) == EXPR) {
            AstNode *expr = node_at(q, -1);

            while (((expr)->_.Expr).rhs && isinstance(((expr)->_.Expr).rhs, EXPR)) {
                expr = ((expr)->_.Expr).rhs;
            }
            if (((expr)->_.Expr).rhs) {
                return 0;
            }
            ((expr)->_.Expr).rhs = inside_brace_node;
        }

        else {
            push_queue(q, inside_brace_node);
        }

        return 1;
    }
    return 0;
}

void push_fun_arg(AstNode *fun, AstNode *arg) {
    if (!isinstance(fun, START_CALL)) {
        ((void)( (!!(0 && "Expected fun node at push_fun_arg")) || (_wassert(L"false && \"Expected fun node at push_fun_arg\"", L".\\ast.c", (unsigned)(514)), 0)) );
    }

    if (((fun)->_.Call).count >= (int)((sizeof(((fun)->_.Call).args)/sizeof(0[((fun)->_.Call).args])) / ((size_t)(!(sizeof(((fun)->_.Call).args) % sizeof(0[((fun)->_.Call).args])))))) {
        ((void)( (!!(0 && "Exceeded limit of fun->args COUNT")) || (_wassert(L"false && \"Exceeded limit of fun->args COUNT\"", L".\\ast.c", (unsigned)(518)), 0)) );
    }

    ((fun)->_.Call).args[((fun)->_.Call).count] = arg;
    ((fun)->_.Call).count += 1;
}


_Bool h_comma_token(Queue *q, Token t, Tokenizer *tokenizer) {
    (void) tokenizer;
    switch (t.type) {
        case T_COMMA: break;
        default: return 0;
    }

    if (node_type_at(q, -2) == START_CALL) {
        AstNode *fun = node_at(q, -2);
        AstNode *arg = pop_queue(q);
        push_fun_arg(fun, arg);
        return 1;
    }

    return 0;
}

AstNode* AstNode_newFunCall(AstNode *start_fun) {
    if (!isinstance(start_fun, START_CALL)) {
        ((void)( (!!(0 && "Expected START_FUN_CALL in call AstNode_newFunCall")) || (_wassert(L"false && \"Expected START_FUN_CALL in call AstNode_newFunCall\"", L".\\ast.c", (unsigned)(545)), 0)) );
    }
    start_fun->type = CALL_EXPR;
    return start_fun;
}

_Bool h_call_start(Queue *q, Token t, Tokenizer *tokenizer) {
    (void) tokenizer;

    switch (t.type) {
        case T_L_BR: break;
        default: return 0;
    }

    if (node_type_at(q, -1) == EXPR) {
        AstNode *prev = node_at(q, -1);

        while (((prev)->_.Expr).rhs && isinstance(((prev)->_.Expr).rhs, EXPR)) {
            prev = ((prev)->_.Expr).rhs;
        }

        if (((prev)->_.Expr).rhs && !isinstance(((prev)->_.Expr).rhs, EXPR)) {

            AstNode* start_fun = AstNode_new(START_CALL, t.tag);
            ((start_fun)->_.Call).callable = ((prev)->_.Expr).rhs;
            ((prev)->_.Expr).rhs = ((void*)0);

            push_queue(q, start_fun);

            return 1;
        }

        return 0;
    }

    if (node_type_at(q, -1) != NONE) {
        AstNode* start_fun = AstNode_new(START_CALL, t.tag);
        AstNode *callable = pop_queue(q);
        ((start_fun)->_.Call).callable = callable;
        push_queue(q, start_fun);
        return 1;
    }

    return 0;
}

_Bool h_call_end(Queue *q, Token t, Tokenizer *tokenizer) {
    (void) tokenizer;

    switch (t.type) {
        case T_R_BR: break;
        default: return 0;
    }

    if (node_type_at(q, -2) == START_CALL || node_type_at(q, -1) == START_CALL) {

        AstNode *start_fun = ((void*)0);
        if (node_type_at(q, -1) != START_CALL) {
            AstNode *last_arg = pop_queue(q);
            start_fun = pop_queue(q);
            push_fun_arg(start_fun, last_arg);
        }
        else {
            start_fun = pop_queue(q);
        }

        AstNode *fun = AstNode_newFunCall(start_fun);

        if (node_type_at(q, -1) == EXPR) {
            AstNode *expr = node_at(q, -1);
            while (((expr)->_.Expr).rhs && isinstance(((expr)->_.Expr).rhs, EXPR)) {
                expr = ((expr)->_.Expr).rhs;
            }
            if (!((expr)->_.Expr).rhs) {
                ((expr)->_.Expr).rhs = fun;
            }
            else {
                push_queue(q, fun);
            }
        }
        else {
            push_queue(q, fun);
        }

        return 1;
    }

    return 0;
}
